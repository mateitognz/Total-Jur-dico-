          }}>

      ))}
    </div>
  );

        </div>
      ))}
    </div>
  );
};

const PipelineView = ({ leads, onLeadClick, onStageChange }) => {

      setDragging(null);
      setHoveredStage(null);
    }
  };

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 20 }}>
      <div>
        <h2 style={{ fontWeight: 900, fontSize: 24, color: COLORS.cream, margin: 0 }}>Pipeline CRM</h2>

      <div style={{ display: "flex", gap: 12, overflowX: "auto", paddingBottom: 8 }}>
        {PIPELINE_STAGES.map((stage) => {
          const stageLeads = leads.filter((l) => l.stage === stage.id);
          const stageValue = stageLeads.reduce((s, l) => s + l.value, 0);

            <div key={stage.id}

                  <span style={{ fontWeight: 700, fontSize: 12, color: COLORS.cream, letterSpacing: 0.3 }}>{stage.label}</span>

              </div>
              {stageValue > 0 && (
                <div style={{ fontSize: 11, color: COLORS.slateLight, marginBottom: 10 }}>{formatCurrency(stageValue)}</div>

                    onMouseEnter={e => e.currentTarget.style.borderColor = stage.color}

                    <div style={{ fontSize: 11, color: COLORS.gold, fontWeight: 600 }}>{lead.radicado}</div>
